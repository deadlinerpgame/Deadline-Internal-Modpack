    recipe Make BMW Classic Engine Guard (with steel bars)
    {
        aerx.IronBar=2,
        aerx.Coke/Base.Charcoal=5,
        Screws=4,
        BlowTorch=5,
        keep WeldingMask,
        keep Wrench,
        keep [Recipe.GetItemTypes.Hammer],
        Result:DL_BikeEngineGuard_BMWClassic,
        OnTest:Recipe.OnTest.DL_BikeEngineGuardAlloy,
        OnCreate:Recipe.OnCreate.DLForge,
        Time:450.0,
        Category:Vehicle Maintenance,
        SkillRequired:MetalWelding=2;Mechanics=6,
        NearItem:Lathe,
        NeedToBeLearn:false,
    }

    fixing Fix BMW Classic Engine Guard Welding
    {

        GlobalItem : BlowTorch=2,
